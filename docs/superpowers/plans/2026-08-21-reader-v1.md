# Reader v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a Windows PySide6 app named Reader that opens `.docx` / `.pptx` / `.xlsx` / `.md` in tabs, with Office COM preview when available and builtin HTML fallback.

**Architecture:** One Python process with a Qt `App` (single-instance `QLocalServer`), one or more `MainWindow`s sharing AppUserModelID `Reader.Desktop`, and a UI-free `PreviewPipeline` that chooses Office export vs format `to_html`. Shell integration writes a desktop shortcut and HKCU “Open with” ProgIDs without touching `UserChoice`.

**Tech Stack:** Python 3.12+, PySide6, pywin32, python-docx, python-pptx, openpyxl, markdown-it-py, pytest, pytest-qt, pytest-mock.

**Spec:** `docs/superpowers/specs/2026-08-21-reader-design.md`

## Global Constraints

- Python 3.12+
- Windows 10/11 only
- Supported extensions (case-insensitive): `.docx`, `.pptx`, `.xlsx`, `.md`
- Reject directories, missing files, `.doc` / `.ppt` / `.xls`, `.pdf`, images, HTML
- Markdown always uses builtin preview (never Office)
- Office COM preferred for Office files; on miss/fail use builtin; status “Office 预览” or “内置预览”
- xlsx builtin: first 1000 rows, truncation notice if more
- Preview cache: `%LOCALAPPDATA%\Reader\preview-cache\`, LRU cap 200 MB
- Single instance; extra launches forward paths via `QLocalServer` name `Reader.SingleInstance.v1`
- AppUserModelID `Reader.Desktop` (one taskbar icon)
- Last tab close leaves an empty window; last window close exits
- Same path in a window focuses the existing tab (no duplicate)
- HKCU Open with only; never write `UserChoice`
- v1 makes no network calls
- Dual pane / convert / translate: stub modules, not wired into UI
- TDD: no production code without a failing test first (except `pyproject.toml` / `.gitignore`)
- Commits: do not skip hooks; do not `git config`

# src/reader/sniff.py
from pathlib import Path

SUPPORTED_EXTENSIONS = frozenset({".docx", ".pptx", ".xlsx", ".md"})


class SniffError(Exception):
    def __init__(self, path: Path, reason: str) -> None:
        self.path = path
        self.reason = reason
        super().__init__(f"{reason}: {path}")


def sniff(path: Path) -> str:
    path = Path(path)
    if not path.exists():
        raise SniffError(path, "not_found")
    if not path.is_file():
        raise SniffError(path, "not_a_file")
    suffix = path.suffix.lower()
    if suffix not in SUPPORTED_EXTENSIONS:
        raise SniffError(path, "unsupported_extension")
    return suffix
```

# src/reader/formats/docx.py
from html import escape
from pathlib import Path
from docx import Document
from reader.preview.result import PreviewResult


def to_html(path: Path) -> PreviewResult:
    doc = Document(str(path))
    parts: list[str] = []
    for p in doc.paragraphs:
        text = escape(p.text)
        if not text:
            continue
        style = (p.style.name or "") if p.style is not None else ""
        if style.startswith("Heading"):
            level = "".join(ch for ch in style if ch.isdigit()) or "1"
            parts.append(f"<h{level}>{text}</h{level}>")
        else:
            parts.append(f"<p>{text}</p>")
    for table in doc.tables:
        rows = ["<table>"]
        for row in table.rows:
            cells = "".join(f"<td>{escape(cell.text)}</td>" for cell in row.cells)
            rows.append(f"<tr>{cells}</tr>")
        rows.append("</table>")
        parts.append("".join(rows))
    body = "\n".join(parts)
    html = f"<!DOCTYPE html><html><head><meta charset='utf-8'></head><body>{body}</body></html>"
    return PreviewResult(html=html, status_label="内置预览", kind="html")
```

# src/reader/formats/pptx.py
from html import escape
from pathlib import Path
from pptx import Presentation
from reader.preview.result import PreviewResult


def to_html(path: Path) -> PreviewResult:
    prs = Presentation(str(path))
    sections: list[str] = []
    for index, slide in enumerate(prs.slides, start=1):
        texts: list[str] = []
        for shape in slide.shapes:
            if not shape.has_text_frame:
                continue
            for para in shape.text_frame.paragraphs:
                t = "".join(run.text for run in para.runs) or para.text
                if t:
                    texts.append(f"<p>{escape(t)}</p>")
            if shape.has_table:
                rows = ["<table>"]
                for row in shape.table.rows:
                    cells = "".join(f"<td>{escape(cell.text)}</td>" for cell in row.cells)
                    rows.append(f"<tr>{cells}</tr>")
                rows.append("</table>")
                texts.append("".join(rows))
        sections.append(
            f'<section class="slide" id="slide-{index}"><h2>Slide {index}</h2>{"".join(texts)}</section>'
        )
    body = "\n".join(sections)
    html = (
        "<!DOCTYPE html><html><head><meta charset='utf-8'>"
        "<style>.slide{border:1px solid #ddd;margin:12px;padding:12px}</style>"
        f"</head><body>{body}</body></html>"
    )
    return PreviewResult(html=html, status_label="内置预览", kind="html")
```

Note: `shape.has_table` may not exist on all python-pptx versions. If the attribute is missing, guard with `try: table = shape.table except Exception: table = None`.

# src/reader/formats/xlsx.py
from html import escape
from pathlib import Path
from openpyxl import load_workbook
from reader.preview.result import PreviewResult


def to_html(path: Path, max_rows: int = 1000) -> PreviewResult:
    wb = load_workbook(str(path), read_only=True, data_only=True)
    nav = ["<nav>"]
    bodies: list[str] = []
    for name in wb.sheetnames:
        nav.append(f'<a href="#sheet-{escape(name)}">{escape(name)}</a> ')
        ws = wb[name]
        rows_out = [f'<section id="sheet-{escape(name)}"><h2>{escape(name)}</h2><table>']
        truncated = False
        data_rows = 0
        for i, row in enumerate(ws.iter_rows(values_only=True)):
            cells = [(c if c is not None else "") for c in row]
            tag = "th" if i == 0 else "td"
            if i > 0:
                data_rows += 1
                if data_rows > max_rows:
                    truncated = True
                    break
            tds = "".join(f"<{tag}>{escape(str(c))}</{tag}>" for c in cells)
            rows_out.append(f"<tr>{tds}</tr>")
        if truncated:
            rows_out.append("<p>truncated</p>")
        rows_out.append("</table></section>")
        bodies.append("".join(rows_out))
    nav.append("</nav>")
    wb.close()
    html = (
        "<!DOCTYPE html><html><head><meta charset='utf-8'></head><body>"
        + "".join(nav)
        + "".join(bodies)
        + "</body></html>"
    )
    return PreviewResult(html=html, status_label="内置预览", kind="html")
```

# tests/test_pipeline.py
from pathlib import Path
import pytest
from reader.preview.result import PreviewResult
from reader.preview.pipeline import preview
from reader.sniff import SniffError

class FakeOffice:
    def __init__(self, available=True, boom=False):
        self.available = available
        self.boom = boom
        self.calls = []

    def available_for(self, suffix: str) -> bool:
        return self.available and suffix in {".docx", ".pptx", ".xlsx"}

    def export(self, path: Path) -> PreviewResult:

    ".docx": ("Word.Application", "Documents"),

    from win32com.client import Dispatch

    Dispatch = None  # type: ignore


def _can_dispatch(progid: str) -> bool:
    if Dispatch is None:
        return False
    try:
        app = Dispatch(progid)
        try:
            app.Quit()
        except Exception:
            pass
        return True
    except Exception:
        return False

    spec = _APPS.get(suffix)
    if spec is None:
        return False
    return _can_dispatch(spec[0])

    def available_for(self, suffix: str) -> bool:
        return office_available(suffix)

    def export(self, path: Path) -> PreviewResult:
        path = Path(path)

        app = Dispatch(progid)
        app.DisplayAlerts = 0

        out_dir = path.parent
        pdf_path = out_dir / (path.stem + ".reader.pdf")

# src/reader/preview/cache.py
from __future__ import annotations
import hashlib
import json
import os
from pathlib import Path
from reader.preview.result import PreviewResult


def cache_dir() -> Path:
    root = Path(os.environ.get("LOCALAPPDATA", Path.home() / "AppData" / "Local"))
    return root / "Reader" / "preview-cache"


class PreviewCache:
    def __init__(self, root: Path | None = None) -> None:
        self.root = Path(root) if root else cache_dir()
        self.root.mkdir(parents=True, exist_ok=True)

    def _key(self, path: Path, strategy: str) -> str:
        st = path.stat()
        payload = f"{path.resolve()}|{st.st_mtime_ns}|{st.st_size}|{strategy}"
        return hashlib.sha256(payload.encode("utf-8")).hexdigest()

    def _slot(self, key: str) -> Path:
        return self.root / key

    def get(self, path: Path, strategy: str) -> PreviewResult | None:
        slot = self._slot(self._key(path, strategy))
        meta = slot / "meta.json"
        if not meta.exists():
            return None
        data = json.loads(meta.read_text(encoding="utf-8"))
        html = (slot / "preview.html").read_text(encoding="utf-8") if data["kind"] == "html" else ""
        pdf = slot / "preview.pdf" if data["kind"] == "pdf" else None
        return PreviewResult(
            html=html,
            status_label=data["status_label"],
            kind=data["kind"],
            pdf_path=pdf if pdf and pdf.exists() else None,
        )

    def put(self, path: Path, strategy: str, result: PreviewResult) -> None:
        slot = self._slot(self._key(path, strategy))
        slot.mkdir(parents=True, exist_ok=True)
        (slot / "meta.json").write_text(
            json.dumps({"status_label": result.status_label, "kind": result.kind}),
            encoding="utf-8",
        )
        if result.kind == "html":
            (slot / "preview.html").write_text(result.html, encoding="utf-8")
        if result.kind == "pdf" and result.pdf_path:
            (slot / "preview.pdf").write_bytes(result.pdf_path.read_bytes())
        self.enforce_limit()

    def enforce_limit(self, max_bytes: int = 200 * 1024 * 1024) -> None:
        entries: list[tuple[float, Path, int]] = []
        total = 0
        for slot in self.root.iterdir():
            if not slot.is_dir():
                continue
            size = 0
            latest = 0.0
            for f in slot.rglob("*"):
                if f.is_file():
                    st = f.stat()
                    size += st.st_size
                    latest = max(latest, st.st_mtime)
            entries.append((latest, slot, size))
            total += size
        entries.sort()
        for _mtime, slot, size in entries:
            if total <= max_bytes:
                break
            for f in slot.rglob("*"):
                if f.is_file():
                    f.unlink()
            slot.rmdir()
            total -= size
```

# src/reader/open.py
from dataclasses import dataclass
from pathlib import Path
from reader.sniff import SniffError, sniff


@dataclass(frozen=True)
class OpenDecision:
    to_open: tuple[Path, ...]
    to_focus: Path | None
    rejected: tuple[tuple[Path, str], ...]


def decide_open(existing: list[Path], incoming: list[Path]) -> OpenDecision:
    existing_res = [Path(p).resolve() for p in existing]
    to_open: list[Path] = []
    rejected: list[tuple[Path, str]] = []
    to_focus: Path | None = None
    for raw in incoming:
        path = Path(raw)
        try:
            sniff(path)
        except SniffError as exc:
            rejected.append((exc.path, exc.reason))
            continue
        resolved = path.resolve()
        if resolved in existing_res:
            to_focus = resolved
            continue
        to_open.append(resolved)
        existing_res.append(resolved)
    return OpenDecision(tuple(to_open), to_focus, tuple(rejected))
```

```python
# src/reader/ipc.py
from __future__ import annotations
import json
from collections.abc import Callable
from PySide6.QtNetwork import QLocalServer, QLocalSocket

SERVER_NAME = "Reader.SingleInstance.v1"


class SingleInstance:
    def __init__(self) -> None:
        self._server: QLocalServer | None = None
        self._on_paths: Callable[[list[str]], None] | None = None

    def become_server(self, on_paths: Callable[[list[str]], None]) -> bool:
        QLocalServer.removeServer(SERVER_NAME)
        server = QLocalServer()
        if not server.listen(SERVER_NAME):
            return False
        self._server = server
        self._on_paths = on_paths
        server.newConnection.connect(self._read_connection)
        return True

    def _read_connection(self) -> None:
        if self._server is None:
            return
        sock = self._server.nextPendingConnection()
        if sock is None:
            return
        sock.waitForReadyRead(1000)
        self._handle_sock(sock)

    def process_once(self) -> None:
        if self._server is None:
            return
        if self._server.waitForNewConnection(1000):
            sock = self._server.nextPendingConnection()
            if sock is not None:
                sock.waitForReadyRead(1000)
                self._handle_sock(sock)

    def _handle_sock(self, sock: QLocalSocket) -> None:
        raw = bytes(sock.readAll()).decode("utf-8")
        paths = json.loads(raw)
        if self._on_paths:
            self._on_paths(paths)
        sock.disconnectFromServer()

    def close(self) -> None:
        if self._server is not None:
            self._server.close()
            QLocalServer.removeServer(SERVER_NAME)
            self._server = None

    @staticmethod
    def send_paths(paths: list[str]) -> bool:
        sock = QLocalSocket()
        sock.connectToServer(SERVER_NAME)
        if not sock.waitForConnected(1000):
            return False
        sock.write(json.dumps(paths).encode("utf-8"))
        sock.flush()
        sock.waitForBytesWritten(1000)
        sock.disconnectFromServer()
        return True
```

from PySide6.QtWidgets import QApplication

APP_USER_MODEL_ID = "Reader.Desktop"


def set_app_user_model_id(app_id: str = APP_USER_MODEL_ID) -> None:
    ctypes.windll.shell32.SetCurrentProcessExplicitAppUserModelID(app_id)


class ReaderApp:
    def __init__(self, qapp: QApplication) -> None:
        self._qapp = qapp
        self._windows: list[MainWindow] = []
        self._ipc = SingleInstance()
        self._ipc.become_server(self._on_ipc_paths)

    def _on_ipc_paths(self, paths: list[str]) -> None:
        win = self._windows[-1] if self._windows else self.new_window()
        win.open_paths(paths)
        win.raise_()
        win.activateWindow()

    def new_window(self) -> MainWindow:
        win = MainWindow(on_new_window=self.new_window)
        win.destroyed.connect(lambda *_: self._drop(win))
        self._windows.append(win)
        win.show()
        return win

    def _drop(self, win: MainWindow) -> None:
        self._windows = [w for w in self._windows if w is not win]

        return len(self._windows)

    def close_all(self) -> None:

from __future__ import annotations
from collections.abc import Callable

from reader.open import decide_open
from reader.preview.office import Win32OfficeBackend
from reader.preview.pipeline import preview


class MainWindow(QMainWindow):
    def __init__(self, on_new_window: Callable[[], MainWindow] | None = None) -> None:
        super().__init__()
        self.setWindowTitle("Reader")
        self._on_new_window = on_new_window
        self._tabs = QTabWidget()
        self._tabs.setTabsClosable(True)
        self._tabs.tabCloseRequested.connect(self.close_tab)
        self.setCentralWidget(self._tabs)
        self.setStatusBar(QStatusBar())
        self.setAcceptDrops(True)
        self._paths: list[Path] = []
        self.actionNewWindow = QAction("新建窗口", self)
        self.actionNewWindow.setObjectName("actionNewWindow")
        self.actionNewWindow.triggered.connect(self._spawn)
        menu = self.menuBar().addMenu("文件")
        menu.addAction(self.actionNewWindow)
        self._office = Win32OfficeBackend()

    def _spawn(self) -> None:
        if self._on_new_window:
            self._on_new_window()

    def tab_count(self) -> int:
        return self._tabs.count()

    def tab_title(self, index: int) -> str:
        return self._tabs.tabText(index)

    def status_text(self) -> str:
        return self.statusBar().currentMessage()

    def focus_path(self) -> str | None:
        idx = self._tabs.currentIndex()
        if idx < 0:
            return None
        return str(self._paths[idx])

    def close_tab(self, index: int) -> None:
        self._tabs.removeTab(index)
        if 0 <= index < len(self._paths):
            self._paths.pop(index)

    def open_paths(self, paths: list[str]) -> None:
        decision = decide_open(self._paths, [Path(p) for p in paths])
        if decision.rejected:
            names = "\n".join(f"{p} ({reason})" for p, reason in decision.rejected)
            QMessageBox.information(self, "Reader", f"无法打开:\n{names}")
        if decision.to_focus is not None:
            self._tabs.setCurrentIndex(self._paths.index(decision.to_focus))
        for path in decision.to_open:
            try:
                result = preview(path, office=self._office)
            except Exception as exc:
                view = QLabel(str(exc))
                self._tabs.addTab(view, path.name)
                self._paths.append(path)
                continue
            if result.kind == "error":
                view: QWidget = QLabel(result.error or "error")
            else:
                web = QWebEngineView()
                if result.kind == "pdf" and result.pdf_path is not None:
                    web.load(result.pdf_path.as_uri())
                else:
                    web.setHtml(result.html)
                view = web
            self._tabs.addTab(view, path.name)
            self._paths.append(path)
            self.statusBar().showMessage(result.status_label)

    def dragEnterEvent(self, event: QDragEnterEvent) -> None:
        if event.mimeData().hasUrls():
            event.acceptProposedAction()

    def dropEvent(self, event: QDropEvent) -> None:
        paths = [u.toLocalFile() for u in event.mimeData().urls()]
        self.open_paths(paths)

# src/reader/shell/associate.py
from __future__ import annotations
import os
from collections.abc import Callable
from pathlib import Path

PROGID = "Reader.Document"
EXTENSIONS = (".docx", ".pptx", ".xlsx", ".md")


def register_open_with(exe: str, winreg_module=None) -> None:
    import winreg as default_wr
    wr = winreg_module or default_wr
    command = f'"{exe}" "%1"'
    cmd_path = r"Software\Classes\Reader.Document\shell\open\command"
    key = wr.CreateKey(wr.HKEY_CURRENT_USER, cmd_path)
    wr.SetValueEx(key, None, 0, wr.REG_SZ, command)
    if hasattr(key, "__enter__"):
        key.__exit__(None, None, None)
    for ext in EXTENSIONS:
        p = rf"Software\Classes\{ext}\OpenWithProgids"
        k = wr.CreateKey(wr.HKEY_CURRENT_USER, p)
        wr.SetValueEx(k, PROGID, 0, wr.REG_SZ, "")
        if hasattr(k, "__enter__"):
            k.__exit__(None, None, None)


def create_desktop_shortcut(
    exe: str,
    name: str = "Reader",
    shortcut_writer: Callable[[str, Path], None] | None = None,
) -> Path:
    dest = Path(os.environ["USERPROFILE"]) / "Desktop" / f"{name}.lnk"
    dest.parent.mkdir(parents=True, exist_ok=True)
    if shortcut_writer is not None:
        shortcut_writer(exe, dest)
        return dest
    from win32com.client import Dispatch
    shell = Dispatch("WScript.Shell")
    sc = shell.CreateShortCut(str(dest))
    sc.Targetpath = exe
    sc.WorkingDirectory = str(Path(exe).parent)
    sc.Description = "Reader"
    sc.save()
    return dest
```

# src/reader/__main__.py
from __future__ import annotations
import sys
from pathlib import Path
from PySide6.QtNetwork import QLocalSocket
from PySide6.QtWidgets import QApplication
from reader.app import ReaderApp, set_app_user_model_id
from reader.ipc import SERVER_NAME, SingleInstance
from reader.shell.associate import create_desktop_shortcut, register_open_with


def _server_absent() -> bool:
    sock = QLocalSocket()
    sock.connectToServer(SERVER_NAME)
    ok = sock.waitForConnected(200)
    sock.disconnectFromServer()
    return not ok


def _launch_target() -> str:
    if getattr(sys, "frozen", False):
        return sys.executable
    return str(Path(__file__).resolve().parents[2] / "scripts" / "reader.cmd")


def main(argv: list[str] | None = None) -> int:
    argv = list(sys.argv if argv is None else argv)
    files = [a for a in argv[1:] if not a.startswith("-")]
    set_app_user_model_id()
    if not _server_absent():
        SingleInstance.send_paths(files)
        return 0
    qapp = QApplication.instance() or QApplication(argv)
    app = ReaderApp(qapp)
    win = app.new_window()
    if files:
        win.open_paths(files)
    try:
        launch = _launch_target()
        register_open_with(launch)
        create_desktop_shortcut(launch)
    except Exception:
        pass
    return qapp.exec()


if __name__ == "__main__":
    raise SystemExit(main())
```
